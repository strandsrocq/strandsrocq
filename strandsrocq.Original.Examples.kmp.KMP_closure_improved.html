Section kmp_closure_improved.

  Definition cl_policy_el__t : Type := (KEY_DATA_T × label__t × KEY_DATA_T).
  Definition cl_policy__t : Type := set cl_policy_el__t.
  Definition reachable_el__t : Type := (KEY_DATA_T × KEY_DATA_T).
  Definition reachable__t : Type := set reachable_el__t.
  Definition closure__t : Type := (cl_policy__t × reachable__t).

  Lemma cl_policy__t_el_eq_dec : ∀ x y : cl_policy_el__t, { x = y } + { x ≠ y }.

  Lemma cl_policy__t_eq_dec : ∀ x y : cl_policy__t, { x = y } + { x ≠ y }.

  Lemma reachable__t_el_eq_dec : ∀ x y : reachable_el__t, { x = y } + { x ≠ y }.

  Lemma reachable__t_eq_dec : ∀ x y : reachable__t, { x = y } + { x ≠ y }.
  Definition reachable__t_leq := incl.

  Lemma closure__t_eq_dec : ∀ x y : closure__t, { x = y } + { x ≠ y }.

  Definition closure__t_leq (C C' : closure__t) := (incl (fst C) (fst C')) ∧ (incl (snd C) (snd C')).

  Notation "C1 '⊴' C2" := (closure__t_leq C1 C2) (at level 40).
  Notation "C '⊢' K '=[' ℓ ']=>' J" := (set_In (K, ℓ, J) (fst C)) (at level 70).
  Notation "C '⊢' K '∈' Z" := (set_In (K, Z) (snd C)) (at level 70).

  Definition is_closure (π : policy__t) (Π : closure__t) : Prop :=
     (∀ KT ℓ JT, π ⊢ KT -[ℓ]-> JT → Π ⊢ (KDn KT) =[ℓ]=> JT) ∧
    (∀ KT, set_In KT (policy_types π) → Π ⊢ KT ∈ KT) ∧
     (Π ⊢ D =[Enc]=> D ∧ Π ⊢ D =[Dec]=> D) ∧
     (∀ K J Z, Π ⊢ K =[Enc]=> J ∧ Π ⊢ K =[Dec]=> Z → Π ⊢ Z ∈ J) ∧
     (∀ K J Z W, Π ⊢ K =[Enc]=> J ∧ Π ⊢ K ∈ Z ∧ Π ⊢ J ∈ W →
                          Π ⊢ Z =[Enc]=> W) ∧
     (∀ K J Z, Π ⊢ K =[Dec]=> J ∧ Π ⊢ K ∈ Z → Π ⊢ Z =[Dec]=> J).
End kmp_closure_improved.
