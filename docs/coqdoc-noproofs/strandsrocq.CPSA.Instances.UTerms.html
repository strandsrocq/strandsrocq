Module Type UTermSig (Import Un : UniverseSig) <: TermSig.

CPSA term algebra

This module signature instantiates the generic strand-space term interface with a CPSA-oriented message algebra. The atom types for names, text, data, tags, symmetric keys, and asymmetric keys are generated from an abstract universe Un with decidable equality.
Messages include typed atoms, encryption, pairing, hashes, and the distinguished void message M_τ used to implement the generic τ element required by the strand-space interface. M_τ is not a basic message and should not be generated by protocol or penetrator behavior.
The term language is intentionally concrete enough for CPSA examples while remaining isolated from the generic bundle and minimality theory. Extending the message constructors should require updates to this instance and its decision procedures, but not to the generic strand-space infrastructure.

  Inductive Name := MkName : U → Name.   Inductive Text := MkText : U → Text.   Inductive Data := MkData : U → Data.   Inductive Tag := MkTag : U → Tag.
  Definition Name_leb a b :=
    match (a, b) with (MkName ua, MkName ub) ⇒ U_leb ua ub end.

  Lemma Name_leb_antisymmetric :
    ∀ a b, Name_leb a b = true → Name_leb b a = true → a = b.

  Lemma Name_leb_total :
    ∀ a b, Name_leb a b = true ∨ Name_leb b a = true.

  Inductive Skey :=
    | skey : U → Skey
    | ltk : Name → Name → Skey.
  Inductive Akey :=
    | akey : U → Akey
    | akey_inv : U → Akey
    | pubk : Name → Akey
    | pubks : Tag → Name → Akey
    | privk : Name → Akey
    | privks : Tag → Name → Akey.
  Definition ainv k := match k with
    | akey a ⇒ akey_inv a
    | akey_inv a ⇒ akey a
    | pubk a ⇒ privk a
    | pubks t a ⇒ privks t a
    | privk a ⇒ pubk a
    | privks t a ⇒ pubks t a
  end.

  Lemma ainv_involutive :
    ∀ k, ainv (ainv k) = k.

  Lemma ainv_injective :
    ∀ k k', ainv k = ainv k' → k = k'.

  Notation "'K' a" := (pubk a)
  (at level 30).
  Definition Ks t a := pubks t a.

  Lemma PK_injective : ∀ A B, K A = K B → A = B.
  Lemma PK_injective_s : ∀ A B t, Ks t A = Ks t B → A = B.

  Definition Bltk a b :=
    if Name_leb a b then ltk a b else ltk b a.


  Lemma SK_bidirectional : ∀ A B, Bltk A B = Bltk B A.

  Lemma SK_injective : ∀ A B A' B',
    Bltk A B = Bltk A' B' → (A = A' ∧ B = B') ∨ (A = B' ∧ B = A').

  Inductive Mesg :=
    
    | M_τ : Mesg
    | M_Name : Name → Mesg
    | M_Text : Text → Mesg
    | M_Data : Data → Mesg
    | M_Tag : Tag → Mesg
    
    | M_Skey : Skey → Mesg
    | M_Akey : Akey → Mesg
    
    | M_encr : Mesg → Mesg → Mesg
    | M_pair : Mesg → Mesg → Mesg
    | M_hash : Mesg → Mesg.

  Definition isBasic m := match m with
    | M_Name _ | M_Text _ | M_Data _ | M_Tag _ | M_Skey _ | M_Akey _ ⇒ True
    | _ ⇒ False
  end.

  Definition Inv m := match m with
    | M_Akey k ⇒ M_Akey (ainv k)
    | _ ⇒ m
  end.

  Lemma Inv_injective : ∀ m m', Inv m = Inv m' → m = m'.
  Lemma Inv_bijective : ∀ m m', Inv m = Inv m' ↔ m = m'.

  #[global] Hint Resolve U_eq_dec : core.

  Notation "k '⁻¹'" := (Inv k) (at level 8).
  Notation "'⟨' m '⟩_' k" := (M_encr k m) (at level 60).
  Notation "a '⋅' b" := (M_pair a b) (at level 60, right associativity).
  Notation "'#' m" := (M_hash m) (at level 10).

  Section Ingredient.

    Fixpoint ingredient (a a' : Mesg) :=
      match a' with
      | M_τ | M_Name _ | M_Text _ | M_Data _ | M_Tag _ | M_Skey _ | M_Akey _ | # _ ⇒ a = a'
      | ⟨g⟩_ _ ⇒ a = a' ∨ ingredient a g
      | g ⋅ h ⇒ a = a' ∨ ingredient a g ∨ ingredient a h
    end.

Subterm is reflexive
    Lemma eq_then_sub : ∀ a, ingredient a a.

  End Ingredient.

  Section TermDecidability.
    Hint Resolve U_eq_dec : core.

    Lemma Mesg_eq_dec : ∀ a a' : Mesg, { a = a' } + { a ≠ a'}.

    Lemma M_ingredient_dec : ∀ a a', { ingredient a a' } + { ~(ingredient a a')}.

    Lemma Name_eq_dec : ∀ t t' : Name, { t = t' } + { t ≠ t' }.

    Lemma Text_eq_dec : ∀ t t' : Text, { t = t' } + { t ≠ t' }.

    Lemma DatM_eq_dec : ∀ t t' : Data, { t = t' } + { t ≠ t' }.

    Lemma Tag_eq_dec : ∀ t t' : Tag, { t = t' } + { t ≠ t' }.

    Lemma Skey_eq_dec : ∀ k k' : Skey, { k = k' } + { k ≠ k' }.

    Lemma Akey_eq_dec : ∀ k k' : Akey, { k = k' } + { k ≠ k' }.

The following lemmas are used in proof automation form LogicalFacts.v that allows for simplifying logical propositions leveraging on decidability.
    Lemma dec_eq_Mesg : ∀ x y: Mesg,
      decidable (x = y).

    Lemma dec_eq_Name : ∀ (x y : Name),
      decidable (x = y).

    Lemma dec_eq_Text : ∀ (x y : Text),
      decidable (x = y).

    Lemma dec_eq_Data : ∀ (x y : Data),
      decidable (x = y).

    Lemma dec_eq_Tag : ∀ (x y : Tag),
      decidable (x = y).

    Lemma dec_eq_Skey : ∀ (x y : Skey),
      decidable (x = y).

    Lemma dec_eq_Akey : ∀ (x y : Akey),
      decidable (x = y).

    Lemma dec_ingredient : ∀ x y,
      decidable (ingredient x y).

  End TermDecidability.

  #[global] Hint Resolve dec_eq_Mesg dec_eq_Name dec_eq_Text dec_eq_Data dec_eq_Tag dec_eq_Skey dec_eq_Akey dec_ingredient : Terms_decidability.
