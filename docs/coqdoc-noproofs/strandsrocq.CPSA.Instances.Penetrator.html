Section PenetratorStrands.

  Variable K__P : Mesg → Prop.

Penetrators (i.e., attackers)
  Inductive penetrator_strand : Σ → Prop :=
    | PT_M : ∀ (m : Mesg) i, isBasic m → K__P m → penetrator_strand (i, [⊕ m])
    | PT_C : ∀ (g h : Mesg) i, penetrator_strand (i, [⊖ g; ⊖ h; ⊕ g⋅h])
    | PT_S : ∀ (g h : Mesg) i, penetrator_strand (i, [⊖ g⋅h; ⊕ g; ⊕ h])
    | PT_E : ∀ (k h : Mesg) i, penetrator_strand (i, [⊖ k; ⊖ h; ⊕ ⟨h⟩_k])
    | PT_D : ∀ (k h : Mesg) i, penetrator_strand (i, [⊖ k⁻¹; ⊖ ⟨h⟩_k; ⊕ h])
    | PT_H : ∀ (m : Mesg) i, penetrator_strand (i, [⊖ m; ⊕ # m]).

  Lemma strand_trace : ∀ i t s, (i,t) = s → t = tr s.
  Definition penetrator_node (n : node__t) := penetrator_strand (strand n).

  Definition never_originates_regular m C :=
    ∀ n, is_node_of n C → originates m n → penetrator_node n.
  Definition originates_regular m C :=
    ∃ n, is_node_of n C ∧ originates m n ∧ ¬ penetrator_node n.

End PenetratorStrands.

Section PenetratorBound.
  Variable B : bundle_type.
  Hypothesis B_is_bundle : is_bundle B.
  Local Notation E := (edges B).

  Proposition penetrator_bound_set :
    ∀ (K__P : Mesg → Prop) (k__R : Mesg),
      isBasic k__R →
      ¬ K__P k__R →
      never_originates_regular K__P k__R B →
      Nsubt B k__R = nil.

  Proposition penetrator_bound :
    ∀ K__P k__R,
      isBasic k__R →
      ¬ K__P k__R →
      never_originates_regular K__P k__R B →
        (∀ p, is_node_of p B → penetrator_node K__P p →
              ¬ (k__R ⊏ (uns_term p))
        ).

  Corollary penetrator_never_learn_secure_decryption_key :
    ∀ K__P s h k k',
      never_originates_regular K__P k'⁻¹ B →
      penetrator_strand K__P s →
      [⊖ k⁻¹; ⊖ ⟨h⟩_k; ⊕ h] = tr s →
      is_node_of (s,0) B →
      isBasic k'⁻¹→
      ¬ K__P k'⁻¹ →
      k ≠ k'.

  Corollary penetrator_never_learn_secure_encryption_key :
    ∀ K__P s h k k',
      never_originates_regular K__P k' B →
      penetrator_strand K__P s →
      [⊖ k; ⊖ h; ⊕ ⟨h⟩_k] = tr s →
      is_node_of (s,0) B →
      isBasic k' →
      ¬ K__P k' →
      k ≠ k'.

End PenetratorBound.
