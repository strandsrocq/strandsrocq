Module Type BundleSizeInductionSig
  (Import T : TermSig)
  (Import St : StrandSig T)
  (Import SSp : StrandSpaceSig T St)
  (Import B : BundleSig T St SSp)
  (Import BI : BundleInductiveSig T St SSp B).

  Definition bundle_size (B : bundle_type) : nat := length (nodes B).

  Theorem bundle_size_induction : ∀ (P : bundle_type → Prop),
      (∀ B n, is_bundle B →
                   (∀ C,
                       is_bundle C →
                       bundle_size C < n →
                       P C) →
                   bundle_size B ≤ n → P B) →
      ∀ n B, is_bundle B → bundle_size B ≤ n → P B.


  Corollary bundle_size_induction_cor1: ∀ (P : bundle_type → Prop),
      (∀ B n, is_bundle B →
                   (∀ C,
                       is_bundle C →
                       bundle_size C < n →
                       P C) →
                   bundle_size B ≤ n → P B) →
      ∀ B, is_bundle B → P B.

  Corollary bundle_size_induction_cor : ∀ (P : bundle_type → Prop),
      (∀ B, is_bundle B →
                   (∀ C,
                       is_bundle C →
                       bundle_size C < bundle_size B →
                       P C) →
                   P B) →
      ∀ B, is_bundle B → P B.

  Definition dedup_bundle (B : bundle_type) : bundle_type :=
    {| nodes := nodup eq_node__t_dec (nodes B);
      intra := nodup eq_edge__t_dec (intra B);
      inter := nodup eq_edge__t_dec (inter B) |}.

  Lemma is_bundle_dedup : ∀ (B : bundle_type),
      is_bundle B → is_bundle (dedup_bundle B).

  Lemma dedup_ordering : ∀ B m n,
      edges B ⊢ m ⪯ n ↔ edges (dedup_bundle B) ⊢ m ⪯ n.

  Lemma dedup_nodes : ∀ B n,
      In n (nodes B) ↔ In n (nodes (dedup_bundle B)).

  Lemma dedup_sink : ∀ B m,
      is_bundle B → sink m B →
      sink m (dedup_bundle B).

  Definition remove_node (m : node__t) (B : bundle_type) :=
    let B' := dedup_bundle B in
    {| nodes := set_remove eq_node__t_dec m (nodes B');
      intra := remove_edges_to m (intra B');
      inter := remove_edges_to m (inter B') |}.

   Lemma remove_sink_bundle : ∀ B m,
      is_bundle B →
      sink m B →
      is_bundle (remove_node m B).

  Lemma length_nodup : ∀ B (l : list B) (eq_dec : ∀ x y, {x=y}+{x≠y}),
      length (nodup eq_dec l) ≤ length l.

  Lemma dedup_le : ∀ B,
      bundle_size (dedup_bundle B) ≤ bundle_size B.

  Lemma NoDup_dedup : ∀ B,
      NoDup (nodes (dedup_bundle B)).

  Lemma remove_sink_smaller : ∀ B m,
      is_bundle B →
      NoDup (nodes B) →
      sink m B →
      bundle_size (remove_node m B) < bundle_size B.

   Lemma remove_sink_dedup_smaller : ∀ B m,
      is_bundle B →
      sink m B →
      bundle_size (remove_node m (dedup_bundle B)) < bundle_size B.


  Corollary bundle_sink_induction : ∀ (P : bundle_type → Prop),
      (∀ B,
          (∀ m, is_bundle B → sink m B →
                     P (remove_node m (dedup_bundle B))) →
          P B) →
      ∀ B, is_bundle B → P B.

  Corollary bundle_sink_induction_alt : ∀ (P : bundle_type → Prop),
      (∀ B,
          (∀ m, NoDup (nodes B) → is_bundle B → sink m B →
                     P (remove_node m B)) →
          P B) →
      ∀ B, NoDup (nodes B) → is_bundle B → P B.


End BundleSizeInductionSig.
