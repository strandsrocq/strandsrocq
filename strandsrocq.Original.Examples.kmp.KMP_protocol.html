  Inductive K_m : K → Prop :=
    | K_master : ∀ u : U, K_m (SymK u).
  Inductive K_d : K → Prop :=
    | K_device : ∀ u : U, K_d (SymK' u).

  Lemma K_m_dec : ∀ k, {K_m k} + {¬K_m k}.

  Lemma K_d_dec : ∀ k, {K_d k} + {¬K_d k}.

  Lemma K_m_then_not_K_d : ∀ k, K_m k → ¬ K_d k.

  Lemma K_d_then_not_K_m : ∀ k, K_d k → ¬ K_m k.

  Definition K__P_md k := match k with
    | SymK u ⇒ False
    | SymK' u ⇒ False
    | _ ⇒ True
  end.

  Lemma K__P_md_then_not_K_d : ∀ k, K__P_md k → ¬ K_d k.
  Lemma K__P_md_then_not_K_m : ∀ k, K__P_md k → ¬ K_m k.
  Lemma K_d_then_not_K__P_md : ∀ k, K_d k → ¬ K__P_md k.
  Lemma K_m_then_not_K__P_md : ∀ k, K_m k → ¬ K__P_md k.

Fresh in the following KMP_strand definition is a declaration, not a property: it has no meaning in this file. A strand space is intended only when Fresh is sound for the bundle under analysis, i.e. satisfies fresh_sound_for below; KMP_secrecy.v and KMP_secrecy_closure_improved.v assume it.
  Definition fresh_sound_for (Fresh : 𝔸 → Prop) (B : bundle_type) : Prop :=
    ∀ t, Fresh t → originates_at_most_once_in B t.

  Inductive KMP_strand (π : policy__t) (Fresh : 𝔸 → Prop) : Σ → Prop :=
  | KMP_C : ∀ (k mk : K) (KT : KEY_T),
      K_d k ∧ K_m mk ∧ Fresh #k ∧ set_In (KDn KT) (policy_types π) →
      ∀ i, KMP_strand π Fresh (i,[⊕ ⟨ #k ⋅ $KT ⟩_mk])
  | KMP_E : ∀ (k mk : K) (KT : KEY_T) (m : K), K_m mk ∧ (π ⊢ KT -[ Enc ]-> D) →
      ∀ i, KMP_strand π Fresh (i,[⊖ #m; ⊖ ⟨ #k ⋅ $KT ⟩_mk; ⊕ ⟨ #m ⟩_k])
  | KMP_D : ∀ (k mk : K) (KT : KEY_T) (m : K), K_m mk ∧ (π ⊢ KT -[ Dec ]-> D) →
      ∀ i, KMP_strand π Fresh (i,[⊖ ⟨ #m ⟩_k; ⊖ ⟨ #k ⋅ $KT ⟩_mk; ⊕ #m])
  | KMP_W : ∀ (k1 k2 mk : K) (KT1 KT2 : KEY_T), K_m mk ∧ (π ⊢ KT2 -[ Enc ]-> KDn KT1) →
      ∀ i, KMP_strand π Fresh (i,[⊖ ⟨ #k1 ⋅ $KT1 ⟩_mk; ⊖ ⟨ #k2 ⋅ $KT2 ⟩_mk; ⊕ ⟨ #k1 ⟩_k2])
  | KMP_U : ∀ (k1 k2 mk : K) (KT1 KT2 : KEY_T), K_m mk ∧ (π ⊢ KT2 -[ Dec ]-> KDn KT1) →
      ∀ i, KMP_strand π Fresh (i,[⊖ ⟨ #k1 ⟩_k2; ⊖ ⟨ #k2 ⋅ $KT2 ⟩_mk; ⊕ ⟨ #k1 ⋅ $KT1 ⟩_mk]).

  Inductive KMP_StrandSpace (π : policy__t) (Fresh : 𝔸 → Prop) : Σ → Prop :=
    | KMPSS_Api : ∀ s, KMP_strand π Fresh s → KMP_StrandSpace π Fresh s
    | KMPSS_Pen : ∀ s, penetrator_strand K__P_md s → KMP_StrandSpace π Fresh s.

  Lemma disjoint_regular_penetrator_strands :
      ∀ π Fresh s, ¬ (KMP_strand π Fresh s ∧ penetrator_strand K__P_md s).

End kmp_protocol.
