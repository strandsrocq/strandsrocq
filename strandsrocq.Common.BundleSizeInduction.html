Library strandsrocq.Common.BundleSizeInduction

From Stdlib Require Import List ListSet Lia Arith.PeanoNat.

From strandsrocq.Common Require Import
  Strands Bundles BundleInductive.

Import ListNotations.
Open Scope list_scope.
Set Implicit Arguments.

Module BundleSizeInduction
  (Import T : TermSig)
  (Import St : StrandSig T)
  (Import SSp : StrandSpaceSig T St)
  (Import B : BundleSig T St SSp)
  (Import BI : BundleInductiveSig T St SSp B).

  Definition bundle_size (b : bundle_type) : nat := length (nodes b).

  Theorem bundle_size_induction : ∀ (P : bundle_type → Prop),
      (∀ b n,
          is_bundle b →
          (∀ c,
              is_bundle c →
              bundle_size c < n →
              P c) →
          bundle_size b ≤ n →
          P b) →
      ∀ n b, is_bundle b → bundle_size b ≤ n → P b.

  Corollary bundle_size_induction_cor : ∀ (P : bundle_type → Prop),
      (∀ b,
          is_bundle b →
          (∀ c,
              is_bundle c →
              bundle_size c < bundle_size b →
              P c) →
          P b) →
      ∀ b, is_bundle b → P b.

  Definition remove_node (m : node__t) (b : bundle_type) : bundle_type :=
    {| nodes := set_remove eq_node__t_dec m (nodes b);
       intra := remove_edges_to m (intra b);
       inter := remove_edges_to m (inter b) |}.

  Lemma remove_sink_bundle : ∀ b m,
      is_bundle b →
      NoDup (nodes b) →
      sink m b →
      is_bundle (remove_node m b).

  Lemma remove_sink_smaller : ∀ b m,
      NoDup (nodes b) →
      sink m b →
      bundle_size (remove_node m b) < bundle_size b.

  Corollary bundle_sink_induction_alt : ∀ (P : bundle_type → Prop),
      (∀ b,
          (∀ m,
              NoDup (nodes b) →
              is_bundle b →
              sink m b →
              P (remove_node m b)) →
          P b) →
      ∀ b, NoDup (nodes b) → is_bundle b → P b.

  Corollary bundle_sink_induction : ∀ (P : bundle_type → Prop),
      (∀ b,
          NoDup (nodes b) →
          is_bundle b →
          (∀ m, sink m b → P (remove_node m b)) →
          P b) →
      ∀ b, NoDup (nodes b) → is_bundle b → P b.

End BundleSizeInduction.
