From Stdlib Require Import Lia.

Require Import Strands.
Require Import LogicalFacts.

Module Type TermTacticsSig.
  Ltac term_tactic_in X Y H := idtac.
  Ltac term_tactic X Y := idtac.
End TermTacticsSig.

Module MakeStrandsTactics
  (Import T : TermSig)
  (Import St : StrandSig T)
  (Import SSp : StrandSpaceSig T St)
  (Import TT : TermTacticsSig).

  Ltac simplify_term :=
    unfold uns_term, uns, term; simpl; try easy;
    match goal with
    | [ H : ?X = tr ?Y |- context [tr ?Y] ] ⇒
      repeat rewrite <-H;
      simplify_term
    | [ H : index ?X = ?Y |- context [index ?X] ] ⇒
      repeat rewrite H;
      simplify_term
    | _ ⇒ auto; idtac
    end.

Ltac simplify_term_in H :=
  unfold uns_term, uns, term in H; simpl in H; try easy;
  match type of H with
  | context [tr ?Y] ⇒
    (match goal with
    | [ H1 : ?X = tr Y |- _ ] ⇒
      repeat rewrite <-H1 in H
    | _ ⇒ fail "no trace"
    end);
    simplify_term_in H
  | context [index ?Y] ⇒
    (match goal with
    | [ H1 : index Y = ?Z |- _ ] ⇒
      repeat rewrite H1 in H
    | _ ⇒ fail "no index"
    end);
    simplify_term_in H
  | _ ⇒ auto; idtac
end.

Lemma or_False_l:
  ∀ P, False ∨ P ↔ P.

Lemma or_False_r:
  ∀ P, P ∨ False ↔ P.

Lemma or_True_l:
  ∀ P, True ∨ P ↔ True.

Lemma or_True_r:
  ∀ P, P ∨ True ↔ True.

Lemma and_True_l:
  ∀ P, True ∧ P ↔ P.

Lemma and_True_r:
  ∀ P, P ∧ True ↔ P.

Lemma and_False_l:
  ∀ P, False ∧ P ↔ False.

Lemma and_False_r:
  ∀ P, P ∧ False ↔ False.

Lemma explosion:
  ∀ P : Prop, (False → P) ↔ True.

Lemma false_implication:
  (True → False) ↔ False.

Lemma notTrue:
  ¬True ↔ False.

Lemma notFalse:
  ¬False ↔ True.

Note this tactic depends on TermTactic X Y in H which is term dependent and is defined in term tactic modules. In case terms are modified it is only required to adapt that part of the tactics.
Ltac simplify_prop_inner H dec :=
  simpl in H;
  unfold not in H;
  repeat (
    match dec with
    | True ⇒
        push not in H using Terms_decidability;
        remove implications in H using Terms_decidability
    | _ ⇒ idtac
    end;
    match type of H with
    | context [?X = ?X] ⇒
      let H1 := fresh "Hassert" in
        assert (X = X ↔ True) as H1 by ( clear; easy );
        rewrite H1 in H;
        clear H1
    | context [?X = ?Y] ⇒
      term_tactic_in X Y H
    | True ⇒ clear H
    | False ⇒ contradiction
    | context [False ∨ ?X] ⇒
      rewrite or_False_l in H
    | context [?X ∨ False] ⇒
      rewrite or_False_r in H
    | context [True ∨ ?X] ⇒
      rewrite or_True_l in H
    | context [?X ∨ True] ⇒
      rewrite or_True_r in H
    | context [True ∧ ?X] ⇒
      rewrite and_True_l in H
    | context [?X ∧ True] ⇒
      rewrite and_True_r in H
    | context [False ∧ ?X] ⇒
      rewrite and_False_l in H
    | context [?X ∧ False] ⇒
      rewrite and_False_r in H
    | context [False → ?X] ⇒
      rewrite explosion in H
    | context [True → False] ⇒
      rewrite false_implication in H
    | context [¬True] ⇒
      rewrite notTrue in H
    | context [¬False] ⇒
      rewrite notFalse in H
    | ?X ∧ ?Y ⇒
      let H1 := fresh "Hand" in
      let H2 := fresh "Hand" in
        destruct H as [H1 H2];
        simplify_prop_inner H1 dec;
        simplify_prop_inner H2 dec
    | ?X ∨ ?Y ⇒
      let H1 := fresh "Hor" in
        destruct H as [H1|H1];
        simplify_prop_inner H1 dec
    | (?X → False) → False ⇒
      try (apply H; clear H; intros H)+(destruct H; intros H; tauto)
    | _ ⇒ idtac
    end
  ).

Tactic Notation "simplify_prop" "in" ident(H) "using" "decidability" :=
    simplify_prop_inner H True;
    subst;
    try easy; (* solve with easy *)
    try auto. (* solve with auto *)

Tactic Notation "simplify_prop" "in" ident(H) :=
  simplify_prop_inner H False;
  subst;
  try easy; (* solve with easy *)
  try auto. (* solve with auto *)

Tactic Notation "simplify_prop" "in" "*" "|-" :=
  repeat (
    match goal with
    | H : _ |- _ ⇒
      progress simplify_prop in H
    end
  ).

  Tactic Notation "simplify_prop" "in" "|-" "*" :=
    simpl;
    unfold not;
    repeat (
      match goal with
      | |- context [?X = ?X] ⇒
        let H1 := fresh "Hassert" in
          assert (X = X ↔ True) as H1 by ( clear; easy );
          rewrite H1;
          clear H1
      | |- context [?X = ?Y] ⇒
        term_tactic X Y
      | |- context [False ∨ ?X] ⇒
        rewrite or_False_l
      | |- context [?X ∨ False] ⇒
        rewrite or_False_r
      | |- context [True ∨ ?X] ⇒
        rewrite or_True_l
      | |- context [?X ∨ True] ⇒
        rewrite or_True_r
      | |- context [True ∧ ?X] ⇒
        rewrite and_True_l
      | |- context [?X ∧ True] ⇒
        rewrite and_True_r
      | |- context [False ∧ ?X] ⇒
        rewrite and_False_l
      | |- context [?X ∧ False] ⇒
        rewrite and_False_r
      | |- context [False → ?X] ⇒
        rewrite explosion
      | |- context [True → False] ⇒
        rewrite false_implication
      | |- context [¬True] ⇒
        rewrite notTrue
      | |- context [¬False] ⇒
        rewrite notFalse
      | |- (?X → False) ⇒
        let H1 := fresh "Hcontra" in
          intros H1;
          simplify_prop in H1
      end
    ).

Tactic Notation "simplify_prop" "in" "*" "|-" "*" :=
  simplify_prop in × |-;
  simplify_prop in |- ×.

Ltac st_implication H :=
  simplify_prop in H;
  try (match type of H with
  | ?X → ?Y ⇒
    let H1 := fresh "Hassert" in
    try (assert (X) as H1 by (
      solve [easy | lia | (simplify_term; try easy; try lia) | intuition]
    );
    specialize (H H1);
    clear H1;
    st_implication H)
  | _ ⇒ idtac
  end).
End MakeStrandsTactics.
