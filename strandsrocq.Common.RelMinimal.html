Library strandsrocq.Common.RelMinimal


From Stdlib Require Import Lists.ListSet.
From Stdlib Require Import Lists.List.
From Stdlib Require Import Relations.

Import Stdlib.Lists.List.ListNotations.

Set Implicit Arguments.

Section RelMinimal.
  Variable A : Set.
  Variable R : A → A → Prop.

  Hypothesis eq_dec : ∀ x y : A, { x=y } + { ¬x=y }.
  Hypothesis R_dec : ∀ x y : A, { R x y } + { ¬R x y }.

  Hypothesis R_antisymmetric: antisymmetric A R.
  Hypothesis R_transitive: transitive A R.

  Notation "n1 '⊑' n2" := (R n1 n2) (at level 60).

  Fixpoint minimal (N : set A) : option A :=
    match N with
    | [] ⇒ None
    | n' :: N' ⇒ match minimal N' with
                | None ⇒ Some n'
                | Some m' ⇒ if R_dec n' m' then Some n' else Some m'
                end
    end.

  Lemma minimal_in_N: ∀ N' m, minimal N' = Some m → set_In m N'.

  Lemma no_minimal_nil: ∀ N, minimal N = None ↔ N = nil.

  Definition is_minimal m N :=
    set_In m N → ∀ n, set_In n N → n≠m → not (n ⊑ m).

  Lemma m_is_minimal : ∀ N m, minimal N = Some m → is_minimal m N.

  Lemma exists_minimal: ∀ N, N ≠ nil → ∃ m, set_In m N ∧ is_minimal m N.
End RelMinimal.
