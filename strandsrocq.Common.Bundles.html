Module Type BundleSig
  (Import T : TermSig)
  (Import St : StrandSig T)
  (Import SSp : StrandSpaceSig T St).

  Module Export BR := BundleRelations T St SSp.

  Record bundle_type := {
    nodes : set node__t;
    intra : edge_set__t;   (* corresponds to n1 ==> n2 *)
    inter : edge_set__t    (* corresponds to n1 --> n2 *)
  }.

  Definition is_node_of n (B : bundle_type) := set_In n (nodes B).
t originates at no more than one node of C: uniqueness without existence. This is the form the development actually uses: every site applies it to two origination proofs to identify the nodes, and none needs a witness. It is also the only one of the two that can be assumed uniformly over a class of terms, since uniquely_originates_in would additionally demand that every term of the class originate in C.
  Definition originates_at_most_once_in C t :=
    ∀ n n', is_node_of n C → is_node_of n' C →
      originates t n → originates t n' → n = n'.
Unique origination as in the strand space papers: t originates in C, and at exactly one node. Prefer originates_at_most_once_in unless a witness is genuinely needed. Keep both relative to C: over all strands the property is unsatisfiable, as (0, [⊕ t]) and (1, [⊕ t]) both originate t.
  Definition uniquely_originates_in C t :=
    (∃ n, is_node_of n C ∧ originates t n) ∧ originates_at_most_once_in C t.
  Definition is_strand_of s (B : bundle_type) := ∀ n,
    strand n = s →
    index n < length (tr (strand n)) →
    is_node_of n B.
  Definition edges (B : bundle_type) := set_union eq_edge__t_dec (intra B) (inter B).
  Definition node_subset_of (N : set node__t) (B : bundle_type) :=
    ∀ n, set_In n N → is_node_of n B.

  Definition is_sub (B : bundle_type) : Prop :=
    
0. edges are resp. subsets of ⟶ and ⟹
    (∀ n1 n2, is_edge_of n1 n2 (intra B) → n1 ⟹ n2) ∧
    (∀ n1 n2, is_edge_of n1 n2 (inter B) → n1 ⟶ n2) ∧
    
0. edges should only relate nodes in nodes B
    (∀ n, incident_to n (intra B) → is_node_of n B) ∧
    (∀ n, incident_to n (inter B) → is_node_of n B).

  Definition is_bundle (B : bundle_type) :=
    
1. B is finite comes for free, set is defined inductively
    is_sub B ∧
    
2. if n1 ∈ nodes B and term(n1) is negative, then there is a unique n2 ∈ nodes B s.t. n2 ⟶ n1 in inter B NOTE: we omit the set_In n2 (nodes B) part since it is implied by is_edge_of n2 n1 (inter B) (see above).
    (∀ n1, set_In n1 (nodes B) → is_negative n1 → ∃! n2, is_edge_of n2 n1 (inter B)) ∧
    
3. if n1 in nodes B and n2 ⟹ n1, then (n2, n1) is in intra B
    (∀ n1 n2, set_In n1 (nodes B) → n2 ⟹ n1 → is_edge_of n2 n1 (intra B)) ∧
    
4. the graph is acyclic, i.e., the transitive closure of edges B is *not* reflexive!
    (∀ n, not (edges B ⊢ n ≺ n)).

  Definition bundle_in_SS (B : bundle_type) (SSp : Σ → Prop) :=
    ∀ n, is_node_of n B → SSp (strand n).
  Definition strandspace_bundle (B : bundle_type) (SSp : Σ → Prop) :=
    is_bundle B ∧ bundle_in_SS B SSp.

  Section BundleProperties.
    Variable B : bundle_type.
    Hypothesis B_is_bundle : is_bundle B.

    Lemma bundle_intrastrand_prefix_closed :
    ∀ n n',
      is_node_of n' B →
      n ⟹+ n' →
      is_node_of n B.

    Corollary index_lt_strand_implies_is_node_of:
      ∀ n n',
        index n < index n' →
        strand n = strand n' →
        is_node_of n' B →
        is_node_of n B.

    Corollary last_node_implies_is_strand_of:
      ∀ n, index n = length (tr (strand n))-1 →
        is_node_of n B →
        is_strand_of (strand n) B.

    Lemma bundle_dec: ∀ n n' B, { is_edge_of n n' B } + { ¬ is_edge_of n n' B }.
  End BundleProperties.

  Section BundleRelationProperties.
    Variable B : bundle_type.
    Hypothesis B_is_bundle : is_bundle B.

    Local Notation E := (edges B).

    Lemma bundle_lt_dec :
      ∀ n1 n2, { E ⊢ n1 ≺ n2 } + { ¬ E ⊢ n1 ≺ n2 }.

    Definition bundle_ltb n n': bool :=
      if bundle_lt_dec n n' then true else false.

    Lemma bundle_ltb_iff_bundle_lt :
      ∀ n n',
      E ⊢ n ≺ n' ↔ bundle_ltb n n' = true.

    Lemma bundle_le_antisymm :
      antisymmetric node__t (bundle_le E).

    Definition partialorder T R := reflexive T R ∧ antisymmetric T R ∧ transitive T R.

    Lemma bundle_le_po : partialorder (bundle_le E).

intrastrand and bundle_le facts
    Lemma intrastrand_implies_bundle_le :
      ∀ n0 n1,
        is_node_of n1 B →
        n0 ⟹+ n1 →
        E ⊢ n0 ⪯ n1.

    Corollary index_le_strand_implies_bundle_le :
      ∀ n0 n1,
        index n0 ≤ index n1 →
        strand n0 = strand n1 →
        is_node_of n1 B →
        E ⊢ n0 ⪯ n1.

    Lemma intrastrand_implies_bundle_lt :
      ∀ n0 n1,
        is_node_of n1 B →
        n0 ⟹+ n1 →
        E ⊢ n0 ≺ n1.

    Corollary index_lt_strand_implies_bundle_lt :
      ∀ n0 n1,
        index n0 < index n1 →
        strand n0 = strand n1 →
        is_node_of n1 B →
        E ⊢ n0 ≺ n1.

  End BundleRelationProperties.

  Section BundleMinimal.
    Variable B : bundle_type.
    Hypothesis B_is_bundle : is_bundle B.
    Local Notation E := (edges B).

    Definition sign_closed (N : set node__t) :=
      node_subset_of N B →
      ∀ m m',
        is_node_of m B →
        is_node_of m' B →
        uns_term m = uns_term m' →
        (set_In m N ↔ set_In m' N).

    Lemma minimal_is_positive :
      ∀ (N : set node__t),
        node_subset_of N B →
        sign_closed N →
        ∀ m, set_In m N →
            is_minimal (bundle_le E) m N →
            is_positive m.

    Definition sign_closed_weak (N : set node__t) :=
      node_subset_of N B →
      ∀ m,
        set_In m N → is_negative m →
        ∃ m', set_In m' N ∧ is_positive m' ∧ E ⊢ m' ≺ m.

    Lemma interstrand_exists_prec_positive_lt_uns:
      ∀ m,
        is_node_of m B → is_negative m →
        ∃ m',
          is_node_of m' B ∧ is_positive m' ∧ E ⊢ m' ≺ m ∧ uns_term m = uns_term m'.

    Lemma sign_closed_implies_weak :
    ∀ N, sign_closed N → sign_closed_weak N.

    Lemma minimal_is_positive_weak :
      ∀ (N : set node__t),
        node_subset_of N B →
        sign_closed_weak N →
        ∀ m, set_In m N →
            is_minimal (bundle_le E) m N →
            is_positive m.

    Definition exists_minimal_bundle :=
      exists_minimal eq_node__t_dec (bundle_le_dec E) (bundle_le_antisymm B_is_bundle) (bundle_le_trans (E:=E)).

  End BundleMinimal.

End BundleSig.
