Section secrecy_guarantee.

  Variable π : policy__t.
  Variable Fresh : 𝔸 → Prop.
  Variable B : bundle_type.
  Local Notation E := (edges B).

  Hypothesis B_is_bundle : is_bundle B.
  Hypothesis B_is_KMP : bundle_in_SS B (KMP_StrandSpace π Fresh).
  Hypothesis fresh_sound : fresh_sound_for Fresh B.
  Fixpoint initial_type_rec N (m : 𝔸) : KEY_DATA_T :=
    match m, N with
    | #k, n::NL ⇒ match tr (strand n) with
                      | [⊕ ⟨ #k' ⋅ ($KT) ⟩_(mk)] ⇒
                        if K_eq_dec k k' then
                          (if K_d_dec k then
                            (if K_m_dec mk then
                              (KDn KT)
                            else initial_type_rec NL m)
                          else initial_type_rec NL m)
                        else initial_type_rec NL m
                      | _ ⇒ initial_type_rec NL m
                      end
    | _, _ ⇒ D
    end.
  Definition initial_type := initial_type_rec (nodes B).

  Lemma initial_type_char :
    ∀ m,
      match m with
      | #k ⇒
        (K_d k →
          ∀ n (KT : KEY_T) mk, K_m mk → is_node_of n B → [⊕ ⟨ #k ⋅ $KT ⟩_(mk)] = tr (strand n) →
            initial_type m = (KDn KT)
        ) ∧
        (¬K_d k → initial_type m = D)
      | _ ⇒ initial_type m = D
      end.

  Lemma initial_type_symm_K_d :
    ∀ k, (K_d k ∧ k = k ⁻¹) ∨ (¬K_d k ∧ initial_type # k = initial_type # k ⁻¹).
  Corollary initial_type_symm :
    ∀ k, initial_type # k = initial_type # k ⁻¹.

  Lemma create_implies_initial_type :
      ∀ k (KT : KEY_T) mk m,
        K_m mk → K_d k → is_node_of m B →
          [⊕ ⟨ #k ⋅ $KT ⟩_(mk)] = tr (strand m) →
            initial_type #k = (KDn KT).

  Variable ϕ : cl_policy__t.
  Variable ℜ : reachable__t.
  Definition ϕ_ℜ_closes_π := is_closure π (ϕ, ℜ).

  Definition wf_cipher_master k KT := (ϕ, ℜ) ⊢ KDn KT ∈ initial_type k.
  Lemma wf_cipher_master_dec : ∀ k KT, {(ϕ, ℜ) ⊢ KDn KT ∈ initial_type k } + {¬(ϕ, ℜ) ⊢ KDn KT ∈ initial_type k}.

  Definition P_wcnmb k k' el :=
    snd (fst el) = Enc ∧ set_In (snd el, initial_type k) ℜ ∧ set_In (fst (fst el), initial_type k') ℜ.

  Lemma P_wcnmb_dec : ∀ k k' x, {P_wcnmb k k' x} + {¬ P_wcnmb k k' x}.

  Definition wf_cipher_nonmaster k k' :=
    ∃ KT KT',
          (ϕ, ℜ) ⊢ KT' =[Enc]=> KT ∧
          ((ϕ, ℜ) ⊢ KT ∈ initial_type k) ∧
          ((ϕ, ℜ) ⊢ KT' ∈ initial_type k').

  Lemma wcnmb_iff_wcnm : ∀ k k', Exists (P_wcnmb k k') ϕ ↔ wf_cipher_nonmaster k k'.

  Lemma wf_cipher_nonmaster_dec : ∀ k k', {wf_cipher_nonmaster k k'} + {¬ wf_cipher_nonmaster k k'}.

  Definition isMK k := match k with | #k' ⇒ K_m k' | _ ⇒ False end.

  Definition isDK k := match k with | #k' ⇒ K_d k' | _ ⇒ False end.

  Lemma isDK_dec : ∀ k, { isDK k } + { ¬isDK k }.

  Fixpoint protected (t : 𝔸) : Prop :=
    match t with
    | ⟨ k ⋅ KT ⟩_(mk) ⇒
      match KT with
        | $KT' ⇒
            (isMK #mk → wf_cipher_master k KT') ∧
            (¬isMK #mk → wf_cipher_nonmaster k #mk ∧ wf_cipher_nonmaster KT #mk) ∧
            (¬isDK k → protected k ∧ protected KT)
        | _ ⇒ protected k ∧ protected KT
      end
    | ⟨ k ⟩_(k') ⇒
        (wf_cipher_nonmaster k #k') ∧ (¬isDK k → protected k)
    | g ⋅ h ⇒ protected g ∧ protected h
    | _ ⇒ (ϕ, ℜ) ⊢ D ∈ (initial_type t)
  end.

  Lemma protected_dec : ∀ t, {protected t} + { ¬protected t}.

  Definition KMP_p (t : 𝔸) : Prop := ¬protected t.

  Lemma KMP_p_dec : ∀ t, { KMP_p t } + { ¬KMP_p t}.

  Definition N_KMP := N KMP_p B KMP_p_dec.
  Definition N_iff_inC_p_KMP := N_iff_inC_p KMP_p B KMP_p_dec.
  Definition minimal_N_KMP_then_mpt := minimal_N_then_mpt KMP_p B_is_bundle KMP_p_dec.

  Lemma no_minimal_is_regular :
    ϕ_ℜ_closes_π →
      ∀ m,
        set_In m N_KMP → is_minimal (bundle_le E) m N_KMP → ¬KMP_strand π Fresh (strand m).

  Lemma KMP_never_originates_mk :
    ∀ k, K_m k → never_originates_regular K__P_md k B.

  Lemma no_minimal_is_penetrator :
    ϕ_ℜ_closes_π →
      ∀ m,
        set_In m N_KMP → is_minimal (bundle_le E) m N_KMP → ¬penetrator_node K__P_md m.

  Lemma N_KMP_no_minimal :
    ϕ_ℜ_closes_π →
      ∀ m, set_In m N_KMP → ¬is_minimal (bundle_le E) m N_KMP.

  Theorem device_key_secrecy:
    ϕ_ℜ_closes_π →
      ∀ m, is_node_of m B → protected (uns_term m).

  Corollary secrecy:
    ϕ_ℜ_closes_π →
      ∀ k m, is_node_of m B →
          K_d k →
            ¬ (ϕ, ℜ) ⊢ D ∈ (initial_type #k) →
              #k ≠ uns_term m.

End secrecy_guarantee.
