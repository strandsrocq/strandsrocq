Module MPT
  (Import T : TermSig)
  (Import St : StrandSig T)
  (Import SSp : StrandSpaceSig T St)
  (Import B : BundleSig T St SSp).

  Section MPT.
    Definition is_mpti n i j α p :=
      (∀ w, w < j-i → ¬p (uns (nth w α (⊖ τ)))) ∧
      i ≤ j ∧
      p (uns (nth (j-i) α (⊖ τ))) ∧
      is_positive_term (nth (j-i) α (⊖ τ)) ∧
      index n = j.

    Definition is_mpt n i α p :=
      ∃ j : nat,
        j < i ∧
        is_mpti n 0 j α p ∧
        index n = j.

    Lemma nth_S_cons_element :
      ∀ (a : sT) α d w, nth (w+1) (a :: α) d = nth w α d.

    Lemma is_mpti_overflow1 :
      ∀ α n i j p, j < i → ¬ is_mpti n i j α p.
    Lemma is_mpti_overflow2 :
      ∀ n i j α p, length α ≤ j-i → ¬ is_mpti n i j α p.
    Lemma is_mpti_head :
      ∀ t α n i j p, j = i → is_mpti n i j (t :: α) p ↔ p (uns t) ∧ is_positive_term t ∧ index n = i.
    Lemma is_mpti_cons :
      ∀ t α n i j p, i < j → is_mpti n i j (t :: α) p ↔ (is_mpti n (i+1) j α p ∧ ¬ p (uns t)).

    Fixpoint mpti (n : node__t) (i : nat) (j : nat) (tr: list sT) (p : A → Prop) : Prop :=
      match tr with
      | [] ⇒ False
      | t :: tr' ⇒
          if (lt_dec i j) then
            ¬ p (uns t) ∧ mpti n (i+1) j tr' p
          else if (eq_dec i j) then
            p (uns t) ∧ is_positive_term t ∧ index n = i
          else False
      end.

    Lemma mpti_overflow1 :
      ∀ α n i j p, j < i → ¬ mpti n i j α p.
    Lemma mpti_overflow2 :
      ∀ α n i j p, length α ≤ j-i → ¬ mpti n i j α p.
    Lemma mpti_head :
      ∀ t α n i j p, j = i → mpti n i j (t :: α) p ↔ (p (uns t) ∧ is_positive_term t ∧ index n = i).
    Lemma mpti_cons :
      ∀ t α n i j p, i < j → mpti n i j (t :: α) p ↔ (mpti n (i+1) j α p ∧ ¬ p (uns t)).

    Lemma is_mpti_iff_mpti:
      ∀ α n i j p,
      is_mpti n i j α p ↔ mpti n i j α p.

    Fixpoint mpt (n : node__t) (i : nat) tr p : Prop :=
      match i with
      | 0 ⇒ False
      | S j ⇒
          mpt n j tr p ∨
          (mpti n 0 j tr p)
      end.

    Lemma is_mpt_then_mpt:
      ∀ n p α i,
      is_mpt n i α p → mpt n i α p.

    Lemma is_mpti_then_is_mpt :
      ∀ n α p,
      is_mpti n 0 (index n) α p →
      is_mpt n (length α) α p.

    Lemma originates_then_is_mpt :
      ∀ t n, originates t n →
      is_mpt n
        (length (tr (strand n)))
        (tr (strand n))
        (fun x ⇒ t ⊏ x).

    Lemma originates_then_mpt:
      ∀ (α:list sT) (n:node__t) (t:A),
        α = tr (strand n) → originates t n →
          mpt n (length α) α (fun x ⇒ t ⊏ x).

    Lemma is_mpti_then_originates:
      ∀ t n,
      is_mpti n 0 (index n) (tr (strand n)) (fun x ⇒ t ⊏ x) →
      originates t n.

    Corollary mpti_then_originates :
      ∀ t n,
      mpti n 0 (index n) (tr (strand n)) (fun x ⇒ t ⊏ x) →
      originates t n.

    Lemma is_mpti_unique :
      ∀ n α p,
      is_mpti n 0 (index n) α p →
      ∀ j, j ≠ index n → ¬ is_mpti n 0 j α p.
  End MPT.

  Section MinimalMPT.
    Import B.
    Variable p : A → Prop.
    Variable B : bundle_type.

    Hypothesis B_is_bundle : is_bundle B.
    Local Notation E := (edges B).
    Hypothesis p_dec : ∀ t, { p t } + { ¬ p t }.

    Definition N := filter (fun n ⇒ if (p_dec (uns_term n)) then true else false) (nodes B).

    Lemma N_iff_inC_p :
      ∀ n, In n N ↔ In n (nodes B) ∧ p (uns_term n).

    Lemma N_is_sign_closed : sign_closed B N.

    Hypothesis np : node__t.
    Definition NW := filter (fun n ⇒ if (p_dec (uns_term n)) then if (bundle_ltb B_is_bundle n np) then true else false else false) (nodes B).
    Lemma NW_iff_inC_p :
      ∀ n, In n NW ↔ In n (nodes B) ∧ p (uns_term n) ∧ E ⊢ n ≺ np.
    Lemma NW_is_weak_sign_closed : sign_closed_weak B NW.

    Lemma minimal_N_then_is_mpti :
      ∀ α n,
      α = tr (strand n) →
      set_In n N → is_minimal (bundle_le E) n N →
      is_mpti n 0 (index n) (α) p.

    Corollary minimal_N_then_mpti :
      ∀ α n,
      α = tr (strand n) →
      set_In n N → is_minimal (bundle_le E) n N →
      mpti n 0 (index n) (α) p.

    Corollary minimal_N_then_is_mpt :
      ∀ α n,
      α = tr (strand n) →
      set_In n N → is_minimal (bundle_le E) n N →
      is_mpt n (length α) α p.

    Corollary minimal_N_then_mpt :
      ∀ α n,
      α = tr (strand n) →
      set_In n N → is_minimal (bundle_le E) n N →
      mpt n (length α) α p.

    Lemma minimal_NW_then_is_mpti :
      ∀ α n,
      α = tr (strand n) →
      set_In n NW → is_minimal (bundle_le E) n NW →
      is_mpti n 0 (index n) (α) p.

    Corollary minimal_NW_then_mpti :
      ∀ α n,
      α = tr (strand n) →
      set_In n NW → is_minimal (bundle_le E) n NW →
      mpti n 0 (index n) (α) p.

    Corollary minimal_NW_then_is_mpt :
      ∀ α n,
      α = tr (strand n) →
      set_In n NW → is_minimal (bundle_le E) n NW →
      is_mpt n (length α) α p.

    Corollary minimal_NW_then_mpt :
      ∀ α n,
      α = tr (strand n) →
      set_In n NW → is_minimal (bundle_le E) n NW →
      mpt n (length α) α p.
  End MinimalMPT.

  Section MinimalOriginatesMPT.
    Import B.

    Variable B : bundle_type.
    Hypothesis B_is_bundle : is_bundle B.
    Local Notation E := (edges B).

    Definition p t x := t ⊏ x.
    Lemma p_dec : ∀ t x : A, { p t x } + { ¬ p t x }.
    Definition Nsubt t := N (p t) B (p_dec t).
    Definition Nsubtiff_inC_p t := N_iff_inC_p (p t) B (p_dec t).
    Definition minimal_Nsubt_then_mpt t := minimal_N_then_mpt (p t) B_is_bundle (p_dec t).
    Definition minimal_Nsubt_then_is_mpti t := minimal_N_then_is_mpti (p t) B_is_bundle (p_dec t).

    Corollary minimal_then_originates :
      ∀ n t,
        In n (Nsubt t) →
        is_minimal (bundle_le E) n (Nsubt t) →
        originates t n.

  End MinimalOriginatesMPT.
End MPT.
