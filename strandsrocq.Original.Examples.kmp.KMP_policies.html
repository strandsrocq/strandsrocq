Section kmp_policies.

  Definition KEY_T := T.
  Inductive KEY_DATA_T := D : KEY_DATA_T | KDn : KEY_T → KEY_DATA_T.
  Inductive label__t := Enc | Dec.

  Coercion KDn : KEY_T >-> KEY_DATA_T.

  Lemma KDn_injective : ∀ K0 K1, KDn K0 = KDn K1 → K0 = K1.

  Definition policy_el__t : Type := (KEY_T × label__t × KEY_DATA_T).
  Definition policy__t : Type := set policy_el__t.

  Lemma KEY_DATA_T_eq_dec : ∀ x y : KEY_DATA_T, { x = y } + { x ≠ y }.

  Lemma policy__t_el_eq_dec : ∀ x y : policy_el__t, { x = y } + { x ≠ y }.

  Lemma policy__t_eq_dec : ∀ x y : policy_el__t, { x = y } + { x ≠ y }.

  Lemma dec_setIn_policy__t : ∀ (x : policy_el__t) y, decidable (set_In x y).
  Definition policy_types (π : policy__t) : set KEY_DATA_T :=
    D :: flat_map (fun e ⇒ match e with (KT, _, JT) ⇒ [KDn KT; JT] end) π.

End kmp_policies.

Notation "π '⊢' K '-[' ℓ ']->' J" := (set_In (K, ℓ, J) π) (at level 70).
#[global] Hint Resolve dec_setIn_policy__t : Terms_decidability.
